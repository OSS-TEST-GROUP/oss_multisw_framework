cmake_minimum_required(VERSION 3.14)
project(sils_vecu CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Phase 1 Vertical PoC (architecture v3 Final, §17 Phase 1 Scope Gate):
# one Contract Catalog endpoint (viss-request, config/sils/contracts/mmf-endpoints.yaml),
# WallClock only, Docker backend, no Relay/lockstep/replay/fault/FMI.
#
# Reuses the project-wide IDL generation from idl/dds/CMakeLists.txt (add_subdirectory(idl)
# runs before add_subdirectory(service) at the top level, so by the time this file
# configures, idl/dds/vehicle_info_mgr/VissRequest.idl has already been generated into
# ${CMAKE_INSTALL_PREFIX}/generated/idl/vehicle_info_mgr/). We do not re-generate or
# duplicate that IDL here -- it is the single Contract Catalog source of truth (§9.1).

if(NOT fastcdr_FOUND)
    find_package(fastcdr 2 REQUIRED)
endif()

if(NOT fastdds_FOUND)
    find_package(fastdds 3 REQUIRED)
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

set(SILS_VISS_GEN_DIR "${CMAKE_INSTALL_PREFIX}/generated/idl/vehicle_info_mgr")
set(SILS_TICK_GEN_DIR "${CMAKE_INSTALL_PREFIX}/generated/idl/sils")

# fastddsgen's exact output filenames vary slightly by version (PubSubTypes /
# TypeObjectSupport pairs); glob rather than hardcode so this doesn't silently
# break on a version bump. Fail loudly at configure time if generation hasn't
# happened yet, instead of a confusing link error later.
file(GLOB SILS_VISS_GENERATED_SRCS "${SILS_VISS_GEN_DIR}/VissRequest*.cxx")
if(NOT SILS_VISS_GENERATED_SRCS)
    message(FATAL_ERROR
        "sils_vecu: no generated sources found under ${SILS_VISS_GEN_DIR}. "
        "Expected idl/dds/vehicle_info_mgr/VissRequest.idl to have been generated "
        "by add_subdirectory(idl) already (top-level CMakeLists.txt adds idl/ before "
        "service/). Check that BUILD_SILS was not configured standalone without the "
        "top-level idl/dds generation step.")
endif()

# TickControl (lockstep clock control plane, §8.2) -- generated from
# idl/dds/sils/TickControl.idl by the same top-level step, into generated/idl/sils.
file(GLOB SILS_TICK_GENERATED_SRCS "${SILS_TICK_GEN_DIR}/TickControl*.cxx")
if(NOT SILS_TICK_GENERATED_SRCS)
    message(FATAL_ERROR
        "sils_vecu: no generated sources under ${SILS_TICK_GEN_DIR}. "
        "Expected idl/dds/sils/TickControl.idl to have been generated by "
        "add_subdirectory(idl). See the VissRequest note above.")
endif()

# fastddsgen can emit duplicate register_*_type_identifier symbols across its
# TypeObjectSupport files when IDLs share a typedef; the SUT hits this and works
# around it the same way (service/core/vehicle_info_mgr/CMakeLists.txt). A single
# IDL (VissRequest) links clean today, so this is preventive parity for when more
# IDLs join the sils build -- harmless on the GNU ld / Linux target.
set(SILS_ALLOW_MULTIDEF -Wl,--allow-multiple-definition)

# Optional FMU plant (§11 L3, FMI 2.0/3.0 co-sim via fmi4c). OFF by default so the
# base build needs no extra dependency; enable with -DSILS_ENABLE_FMU=ON. Uses
# fmi4c (FMI 1/2/3 loader). If fmi4c isn't installed, FetchContent pulls it. When
# enabled, the probe is compiled with SILS_HAVE_FMU and linked against fmi4c so
# FmuPlant (src/plant.hpp) is available (SILS_PLANT=fmu); otherwise SILS_PLANT=fmu
# fails fast at runtime with a clear message.
option(SILS_ENABLE_FMU "Build FmuPlant (FMI co-sim via fmi4c)" OFF)
if(SILS_ENABLE_FMU)
    # Build fmi4c STATIC so the probe has no runtime libfmi4c(d).so dependency (no
    # LD_LIBRARY_PATH juggling at deploy). fmi4c uses its own FMI4C_BUILD_SHARED
    # option (default ON), not BUILD_SHARED_LIBS -- force it off before the fetch.
    # (The FMU's own binary is still dlopen'd from the .fmu at runtime; that's FMI.)
    set(FMI4C_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    find_package(fmi4c QUIET)
    if(NOT fmi4c_FOUND)
        include(FetchContent)
        FetchContent_Declare(fmi4c
            GIT_REPOSITORY https://github.com/robbr48/fmi4c.git
            GIT_TAG main)
        FetchContent_MakeAvailable(fmi4c)
    endif()
endif()

# ── Generated-IDL libraries (T1-B: compile once, link many) ──────────────────
# The fastddsgen output was previously re-globbed and re-compiled into EVERY
# executable and every example (docs/sils-extension-model.md T1-B). Compile each IDL
# ONCE into its own static lib; consumers just link. The include dirs are PUBLIC and
# carry a BUILD/INSTALL split so both the in-tree build and an installed find_package
# consumer resolve the generated headers.
#   - sils_tick_idl : lockstep clock control plane (needed by anything using the tick
#                     loop) -- propagated to all consumers through sils_vecu below.
#   - sils_viss_idl : the VISS wire type -- linked only by VISS-wire consumers
#                     (viss_wire.hpp); a custom-type ECU never links it.
add_library(sils_tick_idl STATIC ${SILS_TICK_GENERATED_SRCS})
target_include_directories(sils_tick_idl PUBLIC
    $<BUILD_INTERFACE:${SILS_TICK_GEN_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/sils_vecu/idl>)
target_link_libraries(sils_tick_idl PUBLIC fastdds fastcdr)
set_target_properties(sils_tick_idl PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(sils_viss_idl STATIC ${SILS_VISS_GENERATED_SRCS})
target_include_directories(sils_viss_idl PUBLIC
    $<BUILD_INTERFACE:${SILS_VISS_GEN_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/sils_vecu/idl>)
target_link_libraries(sils_viss_idl PUBLIC fastdds fastcdr)
set_target_properties(sils_viss_idl PROPERTIES POSITION_INDEPENDENT_CODE ON)

# Canonical vehicle plant-coupling types (MORAI/AWSIM Option 2, bus-coupled external
# world -- docs/sils-morai-plant-adapter-design.md). EgoState (sim -> vECU, STATE) and
# VehicleControl (vECU -> sim, ACTUATION), SI-canonical so the bus contract is sim-
# agnostic. Generated into the same sils dir as TickControl. Linked only by MORAI-mode
# consumers (the channel_map.hpp path); a VISS-only vECU never links it -- same opt-in
# shape as sils_viss_idl.
file(GLOB SILS_VEHICLE_GENERATED_SRCS
    "${SILS_TICK_GEN_DIR}/EgoState*.cxx" "${SILS_TICK_GEN_DIR}/VehicleControl*.cxx"
    "${SILS_TICK_GEN_DIR}/ControlTarget*.cxx")
if(NOT SILS_VEHICLE_GENERATED_SRCS)
    message(FATAL_ERROR
        "sils_vecu: no generated sources for EgoState/VehicleControl/ControlTarget under "
        "${SILS_TICK_GEN_DIR}. Expected idl/dds/sils/{EgoState,VehicleControl,ControlTarget}.idl "
        "to have been generated by add_subdirectory(idl). See the VissRequest note above.")
endif()
add_library(sils_vehicle_idl STATIC ${SILS_VEHICLE_GENERATED_SRCS})
target_include_directories(sils_vehicle_idl PUBLIC
    $<BUILD_INTERFACE:${SILS_TICK_GEN_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/sils_vecu/idl>)
target_link_libraries(sils_vehicle_idl PUBLIC fastdds fastcdr)
set_target_properties(sils_vehicle_idl PROPERTIES POSITION_INDEPENDENT_CODE ON)

# ── libsils_vecu (OSS SDK 경계, P1a) ─────────────────────────────────────────
# 헥사고날 포트(src/*.hpp: IClockSource/IBehavior/IPlant/IScenario/IWireCodec)는
# header-only 라, INTERFACE 라이브러리로 그 include 경로 + DDS 사용요건을 한 handle 로
# 묶는다. 외부 개발자(그리고 아래 실행파일들)가 `target_link_libraries(x sils_vecu)`
# 한 번으로 포트 헤더 + fastdds/fastcdr + TickControl(클록) + FMU 플래그를 얻는다
# (docs/sils-extension-model.md P1). VISS 타입은 sils_vecu 에 없다 — VISS wire 를 쓰는
# 소비자만 sils_viss_idl 을 추가로 link (T1-A: run_loop 는 DDS 타입 무명명).
add_library(sils_vecu INTERFACE)
target_include_directories(sils_vecu INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/sils_vecu>)
target_link_libraries(sils_vecu INTERFACE fastdds fastcdr sils_tick_idl)
target_link_options(sils_vecu INTERFACE ${SILS_ALLOW_MULTIDEF})
if(SILS_ENABLE_FMU)
    target_compile_definitions(sils_vecu INTERFACE SILS_HAVE_FMU)
    target_link_libraries(sils_vecu INTERFACE fmi4c)
endif()
# Namespaced alias so in-tree code and an out-of-tree find_package consumer name the
# target identically (sils::sils_vecu).
add_library(sils::sils_vecu ALIAS sils_vecu)

# The probe/verify/responder/input_source use the VISS wire type (viss_wire.hpp) ->
# link sils_viss_idl. All of them get TickControl + port headers via sils_vecu.
add_executable(sils_vecu_viss_probe src/viss_probe_main.cc)
target_link_libraries(sils_vecu_viss_probe PRIVATE sils_vecu sils_viss_idl)

add_executable(sils_vecu_viss_verify src/viss_verify_main.cc)
target_link_libraries(sils_vecu_viss_verify PRIVATE sils_vecu sils_viss_idl)

# VISS responder (§11 L1) -- SUT stand-in: subscribes /VIN/Vehicle, publishes
# correlated responses on the response topic. VissRequest type only (reused for
# responses via json_min.hpp payload).
add_executable(sils_vecu_viss_responder src/viss_responder_main.cc)
target_link_libraries(sils_vecu_viss_responder PRIVATE sils_vecu sils_viss_idl)

# Lockstep ClockMaster (§8.2) -- independent process, TickControl only (via sils_vecu).
add_executable(sils_clock_master src/sils_clock_master.cc)
target_link_libraries(sils_clock_master PRIVATE sils_vecu)

# Test input source (§7.3 staging fixture) -- publishes async input on an input
# topic; reuses VissRequestTopic (no new IDL).
add_executable(sils_input_source src/sils_input_source.cc)
target_link_libraries(sils_input_source PRIVATE sils_vecu sils_viss_idl)

# MORAI/AWSIM plant-coupling probe (Option 2 S1, docs/sils-morai-plant-adapter-design.md).
# Channel-map driven: reads canonical EgoState off the bus, publishes VehicleControl back
# (free-run). Links sils_vehicle_idl for the canonical types (vehicle_wire.hpp); a separate
# entry point from the VISS probe so that path is untouched.
add_executable(sils_vecu_vehicle_probe src/vehicle_probe_main.cc)
target_link_libraries(sils_vecu_vehicle_probe PRIVATE sils_vecu sils_vehicle_idl)

# EgoState source (test fixture) -- publishes canonical EgoState on the `in` topic so the
# vehicle probe's round trip (state_received -> control_published) can be seen WITHOUT
# MORAI. Stand-in for the sim adapter's STATE broadcast; links sils_vehicle_idl.
add_executable(sils_ego_source src/sils_ego_source.cc)
target_link_libraries(sils_ego_source PRIVATE sils_vecu sils_vehicle_idl)

# ControlTarget source (test fixture) -- publishes canonical ControlTarget on the target
# topic so the Vehicle Interface (VCU) round trip can be seen WITHOUT Autoware. Stand-in for
# the Autoware<->DDS bridge; links sils_vehicle_idl.
add_executable(sils_control_source src/sils_control_source.cc)
target_link_libraries(sils_control_source PRIVATE sils_vecu sils_vehicle_idl)

# Plant-coupling VehicleControl verifier -- independent subscriber on the vECU's OUT channel
# (sils/ego/cmd) that emits verify_pass/verify_fail, giving a bus-coupled run a standalone
# terminal (evaluate.py VERIFIER_TERMINAL) with no coordinator. Bus counterpart of
# sils_vecu_viss_verify; links sils_vehicle_idl for VehicleControl.
add_executable(sils_vecu_vehicle_verify src/sils_vehicle_verify_main.cc)
target_link_libraries(sils_vecu_vehicle_verify PRIVATE sils_vecu sils_vehicle_idl)

# ── Runtime install (always) ─────────────────────────────────────────────────
install(TARGETS sils_vecu_viss_probe sils_vecu_viss_verify sils_vecu_viss_responder
                sils_clock_master sils_input_source sils_vecu_vehicle_probe sils_ego_source
                sils_control_source sils_vecu_vehicle_verify
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

# ── SDK install/export for find_package(sils_vecu) (T1-B Part 2) ─────────────
# An out-of-tree developer repo can `find_package(sils_vecu)` and link sils::sils_vecu
# (+ sils::sils_viss_idl for the VISS wire) instead of vendoring the sources.
#
# Caveat: the FMU-enabled SDK links fmi4c, which is either FetchContent'd (a build target
# not in any export set -> install(EXPORT) fails) or a system package that would need its
# own find_dependency() wiring in the consumer. The base (non-FMU) SDK is the clean export
# target, so we export only then; the FMU build still installs its runtime binaries.
set(SILS_CAN_EXPORT TRUE)
if(SILS_ENABLE_FMU)
    set(SILS_CAN_EXPORT FALSE)
    message(STATUS "sils_vecu: SILS_ENABLE_FMU set -- skipping find_package export "
                   "(fmi4c dependency; runtime binaries still install). Base build exports.")
endif()

if(SILS_CAN_EXPORT)
    install(TARGETS sils_vecu sils_tick_idl sils_viss_idl sils_vehicle_idl
            EXPORT sils_vecuTargets
            ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
            LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
            RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

    # Port headers (src/*.hpp) -- the SDK API surface.
    install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/src/
            DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/sils_vecu
            FILES_MATCHING PATTERN "*.hpp")

    # Generated IDL headers (both types) -- viss_wire.hpp's bare includes resolve here.
    install(DIRECTORY ${SILS_VISS_GEN_DIR}/ ${SILS_TICK_GEN_DIR}/
            DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/sils_vecu/idl
            FILES_MATCHING PATTERN "*.hpp" PATTERN "*.h")

    install(EXPORT sils_vecuTargets
            FILE sils_vecuTargets.cmake
            NAMESPACE sils::
            DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/sils_vecu)

    configure_package_config_file(
        ${CMAKE_CURRENT_SOURCE_DIR}/cmake/sils_vecuConfig.cmake.in
        ${CMAKE_CURRENT_BINARY_DIR}/sils_vecuConfig.cmake
        INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/sils_vecu)
    write_basic_package_version_file(
        ${CMAKE_CURRENT_BINARY_DIR}/sils_vecuConfigVersion.cmake
        VERSION 0.1.0 COMPATIBILITY SameMinorVersion)
    install(FILES
        ${CMAKE_CURRENT_BINARY_DIR}/sils_vecuConfig.cmake
        ${CMAKE_CURRENT_BINARY_DIR}/sils_vecuConfigVersion.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/sils_vecu)
endif()
