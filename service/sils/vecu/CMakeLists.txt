cmake_minimum_required(VERSION 3.14)
project(sils_vecu CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Phase 1 Vertical PoC (architecture v3 Final, §17 Phase 1 Scope Gate):
# one Contract Catalog endpoint (viss-request, config/sils/contracts/mmf-endpoints.yaml),
# WallClock only, Docker backend, no Relay/lockstep/replay/fault/FMI.
#
# Reuses the project-wide IDL generation from idl/dds/CMakeLists.txt (add_subdirectory(idl)
# runs before add_subdirectory(service) at the top level, so by the time this file
# configures, idl/dds/vehicle_info_mgr/VissRequest.idl has already been generated into
# ${CMAKE_INSTALL_PREFIX}/generated/idl/vehicle_info_mgr/). We do not re-generate or
# duplicate that IDL here -- it is the single Contract Catalog source of truth (§9.1).

if(NOT fastcdr_FOUND)
    find_package(fastcdr 2 REQUIRED)
endif()

if(NOT fastdds_FOUND)
    find_package(fastdds 3 REQUIRED)
endif()

set(SILS_VISS_GEN_DIR "${CMAKE_INSTALL_PREFIX}/generated/idl/vehicle_info_mgr")
set(SILS_TICK_GEN_DIR "${CMAKE_INSTALL_PREFIX}/generated/idl/sils")

# fastddsgen's exact output filenames vary slightly by version (PubSubTypes /
# TypeObjectSupport pairs); glob rather than hardcode so this doesn't silently
# break on a version bump. Fail loudly at configure time if generation hasn't
# happened yet, instead of a confusing link error later.
file(GLOB SILS_VISS_GENERATED_SRCS "${SILS_VISS_GEN_DIR}/VissRequest*.cxx")
if(NOT SILS_VISS_GENERATED_SRCS)
    message(FATAL_ERROR
        "sils_vecu: no generated sources found under ${SILS_VISS_GEN_DIR}. "
        "Expected idl/dds/vehicle_info_mgr/VissRequest.idl to have been generated "
        "by add_subdirectory(idl) already (top-level CMakeLists.txt adds idl/ before "
        "service/). Check that BUILD_SILS was not configured standalone without the "
        "top-level idl/dds generation step.")
endif()

# TickControl (lockstep clock control plane, §8.2) -- generated from
# idl/dds/sils/TickControl.idl by the same top-level step, into generated/idl/sils.
file(GLOB SILS_TICK_GENERATED_SRCS "${SILS_TICK_GEN_DIR}/TickControl*.cxx")
if(NOT SILS_TICK_GENERATED_SRCS)
    message(FATAL_ERROR
        "sils_vecu: no generated sources under ${SILS_TICK_GEN_DIR}. "
        "Expected idl/dds/sils/TickControl.idl to have been generated by "
        "add_subdirectory(idl). See the VissRequest note above.")
endif()

# fastddsgen can emit duplicate register_*_type_identifier symbols across its
# TypeObjectSupport files when IDLs share a typedef; the SUT hits this and works
# around it the same way (service/core/vehicle_info_mgr/CMakeLists.txt). A single
# IDL (VissRequest) links clean today, so this is preventive parity for when more
# IDLs join the sils build -- harmless on the GNU ld / Linux target.
set(SILS_ALLOW_MULTIDEF -Wl,--allow-multiple-definition)

# Optional FMU plant (§11 L3, FMI 2.0/3.0 co-sim via fmi4c). OFF by default so the
# base build needs no extra dependency; enable with -DSILS_ENABLE_FMU=ON. Uses
# fmi4c (FMI 1/2/3 loader). If fmi4c isn't installed, FetchContent pulls it. When
# enabled, the probe is compiled with SILS_HAVE_FMU and linked against fmi4c so
# FmuPlant (src/plant.hpp) is available (SILS_PLANT=fmu); otherwise SILS_PLANT=fmu
# fails fast at runtime with a clear message.
option(SILS_ENABLE_FMU "Build FmuPlant (FMI co-sim via fmi4c)" OFF)
if(SILS_ENABLE_FMU)
    find_package(fmi4c QUIET)
    if(NOT fmi4c_FOUND)
        include(FetchContent)
        FetchContent_Declare(fmi4c
            GIT_REPOSITORY https://github.com/robbr48/fmi4c.git
            GIT_TAG main)
        FetchContent_MakeAvailable(fmi4c)
    endif()
endif()

# The probe now drives its tick loop through IClockSource (src/clock_source.hpp),
# whose LockstepClockSource adapter uses the TickControl types -- so the probe
# compiles/links both the VissRequest and TickControl generated sources.
add_executable(sils_vecu_viss_probe
    src/viss_probe_main.cc
    ${SILS_VISS_GENERATED_SRCS}
    ${SILS_TICK_GENERATED_SRCS}
)
target_include_directories(sils_vecu_viss_probe PRIVATE ${SILS_VISS_GEN_DIR} ${SILS_TICK_GEN_DIR})
target_link_libraries(sils_vecu_viss_probe PRIVATE fastdds fastcdr)
target_link_options(sils_vecu_viss_probe PRIVATE ${SILS_ALLOW_MULTIDEF})
if(SILS_ENABLE_FMU)
    target_compile_definitions(sils_vecu_viss_probe PRIVATE SILS_HAVE_FMU)
    target_link_libraries(sils_vecu_viss_probe PRIVATE fmi4c)
endif()

add_executable(sils_vecu_viss_verify
    src/viss_verify_main.cc
    ${SILS_VISS_GENERATED_SRCS}
)
target_include_directories(sils_vecu_viss_verify PRIVATE ${SILS_VISS_GEN_DIR})
target_link_libraries(sils_vecu_viss_verify PRIVATE fastdds fastcdr)
target_link_options(sils_vecu_viss_verify PRIVATE ${SILS_ALLOW_MULTIDEF})

# VISS responder (§11 L1) -- SUT stand-in: subscribes /VIN/Vehicle, publishes
# correlated responses on the response topic. VissRequest type only (reused for
# responses via json_min.hpp payload).
add_executable(sils_vecu_viss_responder
    src/viss_responder_main.cc
    ${SILS_VISS_GENERATED_SRCS}
)
target_include_directories(sils_vecu_viss_responder PRIVATE ${SILS_VISS_GEN_DIR})
target_link_libraries(sils_vecu_viss_responder PRIVATE fastdds fastcdr)
target_link_options(sils_vecu_viss_responder PRIVATE ${SILS_ALLOW_MULTIDEF})

# Lockstep ClockMaster (§8.2) -- independent process, TickControl only.
add_executable(sils_clock_master
    src/sils_clock_master.cc
    ${SILS_TICK_GENERATED_SRCS}
)
target_include_directories(sils_clock_master PRIVATE ${SILS_TICK_GEN_DIR})
target_link_libraries(sils_clock_master PRIVATE fastdds fastcdr)
target_link_options(sils_clock_master PRIVATE ${SILS_ALLOW_MULTIDEF})

# Test input source (§7.3 staging fixture) -- publishes async input on an input
# topic; reuses VissRequestTopic (no new IDL).
add_executable(sils_input_source
    src/sils_input_source.cc
    ${SILS_VISS_GENERATED_SRCS}
)
target_include_directories(sils_input_source PRIVATE ${SILS_VISS_GEN_DIR})
target_link_libraries(sils_input_source PRIVATE fastdds fastcdr)
target_link_options(sils_input_source PRIVATE ${SILS_ALLOW_MULTIDEF})

install(TARGETS sils_vecu_viss_probe sils_vecu_viss_verify sils_vecu_viss_responder
                sils_clock_master sils_input_source
        RUNTIME DESTINATION bin)
